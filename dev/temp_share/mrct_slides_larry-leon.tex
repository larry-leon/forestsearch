\section{Statistical Framework}\label{statistical-framework}

\begin{frame}{Weibull AFT / Cox Dual Representation}
\phantomsection\label{weibull-aft-cox-dual-representation}
\begin{columns}[T]
\begin{column}{0.48\textwidth}

\textbf{AFT Model} (\texttt{survreg}):
$$\log(T) = \log(\theta) + L'\gamma + \tau\varepsilon$$
where $\tau = 1/\nu$ and $\varepsilon \sim$ extreme value.

\medskip

\textbf{Cox Hazard Model}:
$$\lambda(t;\, L) = \lambda_{0}(t)\,\exp(L'\beta)$$
with Weibull baseline $\lambda_{0}(t) = \nu\theta^{-\nu}t^{\nu-1}$.

\end{column}
\begin{column}{0.48\textwidth}

\begin{alertblock}{AFT-to-Hazard Transformation}
$$\beta = -\gamma\,/\,\tau$$
R's \texttt{survreg} estimates $(\log\hat{\theta}, \hat{\gamma}, \hat{\tau})$ on the AFT scale; hazard-ratio coefficients are obtained by dividing by the scale parameter.
\end{alertblock}

\medskip

Potential outcomes are \emph{simulated} on the AFT scale while treatment effects are \emph{interpreted} on the hazard-ratio scale.

\end{column}
\end{columns}
\end{frame}

\begin{frame}[shrink]{Potential Outcomes and Estimands}
\phantomsection\label{potential-outcomes-and-estimands}
\begin{columns}[T]
\begin{column}{0.52\textwidth}

\textbf{Potential Outcome (PO)} $T[x,z]$, under treatment condition $x$ (set to $x=1[0]$ for treatment[control]) and biomarker level $z$


\begin{alertblock}{PO contrast} $\psi^{0}(z) = \{m(0,z) - m(1,z)\}/\tau$: Difference in true means of POs $\log(T[0,z]) - \log(T[1,z])$.
\end{alertblock}

\textbf{Strong prognostic $W$:} Linear predictor $l_{x,z,w} = \beta_{1}^{0}x + \beta_{2}^{0}z + \beta_{3}^{0}zx + \beta_{4}^{0}(z-k)I(z>k) + \beta_{5}^{0}(z-k)I(z>k)x + \beta_{w}^{0}w$

\begin{alertblock}{PO hazard function (omitting baseline)}
$\theta^{x}(z,v) = \exp(l(x,z,v))$
\end{alertblock}

\end{column}
\begin{column}{0.43\textwidth}

\begin{alertblock}{Average Hazard Ratio (AHR)}
$$\text{AHR}(z^{+}) = \exp\!\bigl\{E_{Z \geq z}\;\psi^{0}(Z)\bigr\}$$
Exponentiated average of differences in true means of POs (causal log-hazard ratios across $z$)
\end{alertblock}

\begin{alertblock}{Controlled Direct Effect (CDE)}
 Let $\bar{\theta}^{x}(z+) = E_{Z \geq z,V}\theta^{x}(Z,V)$


$$\text{CDE}(z^{+}) = \bar{\theta}^{1}(z+)\;/\;\bar{\theta}^{0}(z+)$$

Ratio of average exponentiated hazards
\end{alertblock}

\end{column}
\end{columns}
\end{frame}

\begin{frame}[shrink]{Subgroup Defined by (general) \(H\)}
\phantomsection\label{subgroup-defined-by-general-h}
\begin{columns}[T]
\begin{column}{0.48\textwidth}

\begin{alertblock}{AFT Model Specification}
$$\log T_i(x=1) = \mu + \mathbf{L}_i(1)^{\top} \boldsymbol{\gamma} + \tau\,\varepsilon_i$$
\end{alertblock}

The covariate vector includes treatment, prognostic effects ($V$, say), and an interaction:
$$\mathbf{L}_i(x=a)^{\top}\boldsymbol{\gamma} = a \cdot \gamma_{\text{treat}} + \mathbf{V}_i^{\top}\boldsymbol{\gamma}_V + a \cdot H_i \cdot \gamma_{\text{int}}$$
where $H_i \in \{0,1\}$ flags harm-subgroup membership ($\mathcal{S}$).

$$\text{AHR}(\mathcal{S}) = \exp\!\left\{\frac{1}{|\mathcal{S}|}\sum_{i \in \mathcal{S}} \psi_i^{0}\right\}$$

\end{column}

\begin{column}{0.48\textwidth}

The \textbf{marginal hazard ratio} for a subgroup $\mathcal{S}$ is obtained from a Cox model fit to stacked potential outcomes:
$\text{HR}_{\text{marg}}(\mathcal{S}) = \exp\!\bigl(\hat{\beta}_{\text{Cox}} \;\big|\; \text{data} = \{(T_i(1), 1, 1),\, (T_i(0), 1, 0) : i \in \mathcal{S}\}\bigr)$

\textbf{Computation in ForestSearch}:
\begin{enumerate}
\item Generate shared error: $\varepsilon_i = \log(\text{Exp}(1))$
\item Compute potential survival times:
\begin{align*}
\log T_i(1) &= \mu + \sigma\varepsilon_i + \mathbf{L}_i(1)^{\top}\boldsymbol{\gamma} \\
\log T_i(0) &= \mu + \sigma\varepsilon_i + \mathbf{L}_i(0)^{\top}\boldsymbol{\gamma}
\end{align*}
\item Fit: \texttt{coxph(Surv(time, event) \textasciitilde{} treat)}
\item Extract: $\text{HR}_{\text{marg}} = \exp(\hat{\beta}_{\text{treat}})$
\end{enumerate}

\end{column}
\end{columns}
\end{frame}

\begin{frame}{ForestSearch Algorithm (León et al., 2024, \emph{SIM})}
\phantomsection\label{forestsearch-algorithm-leuxf3n-et-al.-2024}
\begin{enumerate}
\item \textbf{Candidate Selection} --- From baseline covariates to $L$ candidate subgroup indicators
  \begin{itemize}
  \item \textit{LASSO}: Cox regularization for prognostic selection; mitigates false discovery from noise
  \item \textit{GRF}: Causal survival forest targets RMST to identify treatment effect heterogeneity
  \item Continuous covariates $\rightarrow$ binary splits (medians, quartiles, GRF cuts)
  \end{itemize}

\item \textbf{Exhaustive Subgroup Search} --- All combos of $L$ candidates (default max of 2-way)
  \begin{itemize}
  \item $L(L-1)/2 + L$ possible combinations from $L$ subgroup indicators (max = 2)
  \item Filter: $n \geq 60$, events $d_0 \geq 12$, and $d_1 \geq 12$
  \item Screen: Cox HR $\geq 1.25$ (experimental vs.\ control) ``signal of harm''
  \end{itemize}

\item \textbf{Splitting Consistency} --- Validate each candidate ($R = 1,000$ splits)
  \begin{itemize}
  \item Randomly split subgroup 50/50; require \emph{both halves} HR $\geq 1.0$ ``consistent with harm''
  \item $\hat{p}_{\text{cons}} = R^{-1}\sum_{r} I(\min(\hat{\beta}_s^{1r}, \hat{\beta}_s^{2r}) \geq 0) \geq 90\%$
  \item Select $\widehat{H}$: subgroup ``maximally consistent with harm''
  \end{itemize}
\end{enumerate}
\end{frame}

\section{Key Takeaways}\label{key-takeaways}

\begin{frame}{Summary}
\phantomsection\label{summary}
\begin{enumerate}
\item
  \textbf{ForestSearch} provides a principled approach to subgroup
  identification that directly addresses MRCT regional consistency
  challenges
\item
  The \textbf{Weibull AFT/Cox dual framework} enables causal
  interpretation --- simulated on AFT scale, interpreted via hazard
  ratios
\item
  Three complementary estimands (\textbf{AHR}, \textbf{CDE},
  \textbf{Marginal HR}) robustly characterize treatment effect
  heterogeneity
\item
  The \textbf{train/test paradigm} (non-AP \(\to\) AP) avoids
  overfitting
\item
  \textbf{Data-driven simulation} evaluates full operating
  characteristics --- false discovery, estimation bias, CI coverage ---
  grounded in real trial data
\item
  \textbf{Simulation results} demonstrate the approach can recover AP
  consistency under heterogeneous effects while controlling
  false-discovery under the null
\end{enumerate}
\end{frame}

\section{Additional Notes}\label{additional-notes}

\begin{frame}[shrink]{Some More Details}
\phantomsection\label{some-more-details}
\begin{columns}[T]
\begin{column}{0.48\textwidth}

\textbf{Dual Causal Status}

The Weibull AFT scale-change parameter $\exp(-\beta)$ equals the \textbf{geometric mean ratio} of potential survival times:
$$\exp\!\bigl(E[\log T_{i}(1) - \log T_{i}(0)]\bigr)$$
Simultaneously \textbf{individual-level} and \textbf{population-level} causal.

\medskip

\textbf{AHR Inherits This Status}:
$$\text{AHR}(\mathcal{S}) = \exp\!\left(\frac{1}{|\mathcal{S}|}\sum_{i \in \mathcal{S}} [\theta_{i}(1) - \theta_{i}(0)]\right)$$

\end{column}
\begin{column}{0.48\textwidth}

\textbf{Cox HR as Operational Tool}

\begin{itemize}
\item \emph{True} effects: defined through $\psi^{0}(z)$
\item Subgroup \emph{selection}: Cox HR as efficient screening
\item Subgroup \emph{evaluation}: AHR and CDE
\item \emph{Correction}: bootstrap bias correction (\texttt{hr.H.bc})
\end{itemize}

\medskip

\textbf{De-Biased Estimation}:
$$\hat{\beta}^{*}(\hat{\mathcal{H}}) = \hat{\beta}(\hat{\mathcal{H}}) - \frac{1}{B}\sum_{b=1}^{B}\bigl[\eta^{*}_{b}(\hat{\mathcal{H}}^{*}_{b}) + \eta^{*}_{b}(\hat{\mathcal{H}})\bigr]$$

Infinitesimal jackknife for valid CIs.

\end{column}
\end{columns}
\end{frame}

\begin{frame}[shrink]{ForestSearch Implementation Details}
\phantomsection\label{forestsearch-implementation-details}
\begin{columns}[T]
\begin{column}{0.48\textwidth}

\textbf{Feature Discovery Pipeline}

\begin{enumerate}
\item \textbf{GRF} (causal survival forests): treatment effect heterogeneity via RMST
\item \textbf{LASSO} regularization: variable screening via Cox penalized regression
\item \textbf{Exhaustive combinatorial search}: across factors and cutpoints
\item \textbf{Forced cutpoints}: \texttt{conf\_force} for domain-guided biomarker cuts
\end{enumerate}

\medskip

\textbf{Selection Criteria}

\begin{itemize}
\item HR threshold: $\hat{\beta} \geq \log(0.90)$ 
\item Consistency rate $\geq$ 80--90\% 
\item Minimum sizes: $n \geq 60$, $d \geq 12$ 
\item \texttt{sg\_focus}: \texttt{"minSG"} $|$ \texttt{"hr"} $|$ \texttt{"maxSG"}
\end{itemize}

\end{column}
\begin{column}{0.48\textwidth}

\textbf{Cross-Validation}
\begin{itemize}
\item \texttt{leave-one-out}
\item \texttt{10-fold}
\item \texttt{agreement metrics: cv-testing vs full} sens, ppv, subgroup match (cv-training vs full) metrics
\end{itemize}

\medskip
\textbf{Bootstrap Inference}
\begin{itemize}
\item parallelised (dofuture) bootstrap
\item Full algorithm mimicked in each replicate ($B = 300$--$2000$)
\item IJ variance estimation for valid confidence intervals
\end{itemize}
