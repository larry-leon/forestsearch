\section{Motivation \& Overview}\label{motivation-overview}

\begin{frame}{The Core Question}
\phantomsection\label{the-core-question}
\begin{block}{Subgroup analysis in regulatory submissions}
When many subgroups are examined in a single trial, small subgroups can
produce hazard ratio estimates that look dramatic \emph{purely by chance}.
\end{block}

\begin{alertblock}{This study asks directly}
Under a DGM with a \textbf{perfectly uniform treatment benefit}
(true HR\,=\,0.70 for every patient), how often do standard Cox model
analyses produce extreme subgroup results?
\end{alertblock}

\begin{itemize}
\tightlist
\item
  Clinical subgroups of a given \(N\) are statistically
  indistinguishable from meaningless \emph{random} groupings of the same
  size
\item
  The \textcolor{randomorange}{\textbf{random}} benchmarks random
  combinations and should fluctuate around overall results with
  ``extremeness'' due to variability of small samples as opposed to true
  subgroup effects
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Setting: Uniform Treatment Effect}
\phantomsection\label{setting-uniform-treatment-effect}
The DGM is built from a real trial (GBSG breast cancer, \(N = 686\))
using \texttt{generate\_aft\_dgm\_flex()}.

\begin{block}{Key feature --- constant log hazard ratio}
$$\psi^{0}(\mathbf{L}) = \log(\mathrm{HR}_{\mathrm{true}}) \quad \text{for all } \mathbf{L}.$$
\end{block}

\begin{itemize}
\tightlist
\item
  True overall HR = 0.70 (moderate benefit)
\item
  Every subgroup shares \emph{exactly} this HR
\item
  Any apparent subgroup finding is a false positive by construction
\end{itemize}

\begin{exampleblock}{Spline parameterisation}
\texttt{log.hrs = log(c(0.70, 0.70, 0.70))} --- uniform HR at biomarker
values $0$, knot $k$, and upper anchor $\zeta$.
\end{exampleblock}
\end{frame}

\section{Part I --- Building the Null
DGM}\label{part-i-building-the-null-dgm}

\begin{frame}[fragile]{Fitting \texttt{generate\_aft\_dgm\_flex()}}
\phantomsection\label{fitting-generate_aft_dgm_flex}
\begin{itemize}
\tightlist
\item
  Fits a \textbf{Weibull AFT model} to source data (GBSG)
\item
  Generates a large \emph{super population}
  (\(N_{\text{super}} = 5000\)) from which all future trials are drawn
\item
  \texttt{model\ =\ "null"} --- treatment effect identical for every
  patient
\item
  Censoring model estimated by AIC selection (Weibull / log-normal)
\end{itemize}

\begin{exampleblock}{R code}
\small\ttfamily
dgm\_null <- generate\_aft\_dgm\_flex(\\
\quad data = gbsg, model = "null",\\
\quad subgroup\_vars = NULL, ~~ n\_super = 5000)
\end{exampleblock}
\end{frame}

\section{Part II --- Censoring Diagnostics \&
Calibration}\label{part-ii-censoring-diagnostics-calibration}

\begin{frame}[fragile]{Censoring Consistency}
\phantomsection\label{censoring-consistency}
\begin{itemize}
\tightlist
\item
  Credible results require censoring in synthetic trials to match source
  data
\item
  \texttt{check\_censoring\_dgm()} compares three quantities:

  \begin{enumerate}
  \tightlist
  \item
    \textbf{Overall censoring rate} vs \texttt{rate\_tol} (default 10
    pp)
  \item
    \textbf{Censoring-time quantiles} among censored subjects
  \item
    \textbf{KM median censoring time} vs \texttt{median\_tol} (default
    25\%)
  \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Calibrating \texttt{cens\_adjust}}
\phantomsection\label{calibrating-cens_adjust}
\texttt{calibrate\_cens\_adjust()} uses \texttt{uniroot} to find the
log-scale shift that equates a target statistic between reference and
simulated data.

\begin{block}{Objective function}
At each candidate \texttt{cens\_adjust}:
(1) generate synthetic subjects,
(2) extract target metric,
(3) return sim\_metric $-$ ref\_metric.
\end{block}

\begin{center}
\begin{tabular}{ll}
\toprule
Target & Equates \\
\midrule
\texttt{"rate"} & Overall censoring rate \\
\texttt{"km\_median"} & KM-based median censoring time \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Monotonicity:} larger \texttt{cens\_adjust} \(\Rightarrow\)
longer censoring times.
\end{frame}

\section{Part III --- Generating Synthetic
Trials}\label{part-iii-generating-synthetic-trials}

\begin{frame}[fragile]{\texttt{simulate\_from\_dgm()}}
\phantomsection\label{simulate_from_dgm}
With the null DGM calibrated, each simulated trial is a single function
call:

\begin{enumerate}
\tightlist
\item
  Samples from super population with replacement
\item
  Assigns treatment (\(1{:}1\) randomisation)
\item
  Generates latent survival times from the AFT model
\item
  Applies calibrated censoring model
\item
  Truncates at \texttt{analysis\_time}
\end{enumerate}

Under the null DGM: \texttt{flag\_harm} \(\equiv\) 0 for every subject.
\end{frame}

\section{Key Findings}\label{key-findings}

\begin{frame}{HR Point Estimates}
\phantomsection\label{hr-point-estimates}
\begin{block}{Large subgroups ($N > 200$)}
Median HR tracks truth ($\approx 0.70$) closely; ECI is narrow.
\end{block}

\begin{alertblock}{Small subgroups ($N \lesssim 40$)}
\begin{itemize}
\item ECI for \texttt{random15}/\texttt{random20} spans $\sim\!0.25$ to $>1.0$
\item Clinical subgroups of similar size show \textbf{indistinguishable} ECI widths
\item Clinical rationale provides \textbf{no statistical protection}
\end{itemize}
\end{alertblock}

\begin{center}
\fcolorbox{UBCblue}{UBCblue!5}{\parbox{0.85\textwidth}{\centering
\textbf{Clinical subgroups of a given $N$ are statistically
indistinguishable from meaningless random groupings of the same size.}
}}
\end{center}
\end{frame}

\begin{frame}{Upper Bounds UB(HR)}
\phantomsection\label{upper-bounds-ubhr}
\begin{columns}[T]
\begin{column}{0.55\linewidth}
\begin{block}{The upper-bound distribution}
\begin{itemize}\small
\item \texttt{random40} ($N\!\approx\!40$): 1\% of trials produce UB(HR)\,$>$\,3.0--4.0
\item \texttt{random15} ($N\!\approx\!15$): values above 5.0 not rare
\item A reviewer seeing UB(HR)\,=\,3.5 in 15 patients cannot distinguish
      chance from a genuine safety signal
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.42\linewidth}
\begin{exampleblock}{\footnotesize Illustrative 99th pctl of UB(HR)}

$N\!\approx\!200$ & $\sim\!1.5$ \\
$N\!\approx\!60$ & $\sim\!2.5$ \\
random40 & $\sim\!3.5$ \\
random20 & $\sim\!5.0$ \\

\bottomrule
\end{tabular}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Cox Model Works Correctly}
\phantomsection\label{cox-model-works-correctly}
\begin{block}{What the simulation confirms}
\begin{enumerate}
\item Cox model is \textbf{unbiased} --- median HR $\approx$ 0.70 for every subgroup
\item Confidence intervals have correct \textbf{coverage}
\item Variability driven \textbf{entirely by sample size}
\end{enumerate}
\end{block}

\begin{alertblock}{The interpretive hazard}
\begin{itemize}
\item UB(HR)\,=\,3.5 in a 15-patient subgroup is \emph{expected} under uniform benefit
\item Without simulation-based calibration, no way to quantify how ``extreme'' a result is
\item The \texttt{random*} benchmarks provide exactly this calibration
\end{itemize}
\end{alertblock}
\end{frame}

\section{Conclusions}\label{conclusions}

\begin{frame}[fragile]{Conclusions}
\phantomsection\label{conclusions-1}
\begin{enumerate}
\item
  \textbf{Small subgroups produce extreme results by chance.} Under
  uniform HR = 0.70, subgroups with \(N \lesssim 40\) regularly produce
  UB(HR) \(> 3.0\).
\item
  \textbf{Clinical rationale does not protect against noise.} Clinical
  subgroups have the \emph{same} HR/UB distribution as random benchmarks
  of that size.
\item
  \textbf{The Cox model is working correctly.} Wide CIs in small
  subgroups are accurate --- they reflect genuine uncertainty.
\item
  \textbf{Simulation-based calibration is essential.} Without a null-DGM
  benchmark, reviewers cannot quantify ``extreme'' for a given subgroup
  size.
\item
  \textbf{The \texttt{forestsearch} package} provides a complete,
  reproducible workflow.
\end{enumerate}
\end{frame}
